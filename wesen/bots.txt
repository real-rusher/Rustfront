# DUSTFRONT - die Bots des Mehrspielers als zielorientierte KI (seit 0.34.29)
#
# Gilt mit der Regel BOT-KI: GOAP (Rundentafel, ERWEITERT) und im Testmodus
# (Spawner, BOT, HIRN: GOAP). Die klassische KI (bots.py) bleibt daneben,
# bis diese hier besser ist. Format: siehe dustfront/wesen_daten.py,
# Planung: docs/KI.md.
#
# Fakten, die ein Bot ueber sich weiss (bot_goap.GoapHirn.fuehlen):
#   feind_sichtbar       er sieht einen Gegner auf seiner Ebene
#   linie_frei           ... und zwischen ihnen haelt nichts die Kugel auf
#   feind_bekannt        er sieht einen oder hat eben noch einen gesehen
#   feind_gehoert        er hoert einen, den er nicht sieht (seit 0.34.38)
#   munition_da          irgendeine Waffe hat noch Munition
#   magazin_leer         das Magazin der Waffe in der Hand ist leer (oder er laedt)
#   granate_da           er hat seine Sprenggranate noch
#   angeschlagen         weniger als K.BOTS["deckung_leben"] Leben
#   getroffen            in den letzten zwei Sekunden getroffen
#   unterdrueckt         Kugeln pfeifen knapp an ihm vorbei
#   in_deckung           er steht an einer Stelle, die ihn vom Gegner trennt
#   beruhigt             sein Stress ist unter 0,3 (seit 0.34.38)
#   kamerad_liegt        ein Kamerad liegt am Boden (VERSUS)
#   kamerad_braucht_schutz  ein Kamerad laedt, ist angeschlagen, liegt oder flankiert
#   kamerad_flankiert    ein anderer flankiert den Gegner, den er kennt
#   flanke_frei          er darf flankieren (keiner sonst, keine Pause)
#   zone_aktiv           es gibt einen Kreis (HUEGEL)
#   im_kreis             er steht darin
#   sturm_plan           er ist Teil eines Sturms seiner Mannschaft
# Seit 0.34.38 (innenleben.py, docs/KI.md 8.3):
#   granate_nah          eine scharfe Granate liegt, die ihn erreicht
#   panik                er ist in Panik
#   eingeschossen        er feuert schon eine Weile ohne Pause
#   waffe_mg, waffe_auto er haelt das MG / eine Waffe fuer Dauerfeuer
#   feinde_gruppe        der Gegner, den er sieht, steht mit anderen beisammen
#   ziel_fast_tot        der Gegner, den er sieht, hat kaum noch Leben
#   flankiert            Gegner von zwei Seiten (mehr als 80 Grad), und er ist nicht vor jedem gedeckt (0.34.50)
# Seit 0.34.39, im Trupp (trupp.py) - ohne Trupp oder als Letzter alle falsch:
#   im_trupp             er ist in einem Trupp mit wenigstens einem anderen
#   ist_fuehrer          er fuehrt ihn
#   hat_funk             er hat ein Funkgeraet (nur der Anfuehrer, Testmodus)
#   allein               keiner aus dem Trupp in Sprechweite
#   in_formation         er steht an seiner Stellung (Bahn, Platz in der Linie)
#   trupp_rueckzug, trupp_druck, trupp_halten   was der Trupp gerade tut
#   welle_dran           er zieht in dieser Welle (sonst deckt er)
#   flanke_rolle         er flankiert fuer den Trupp
#   trupp_unterlegen     der Trupp ist besorgt oder in der Unterzahl
#   hilfe_gerufen        der Trupp hat eben um Hilfe gefunkt
#   lage_neu             es gibt Neues zu melden
#   antwort_offen        ein anderer rief um Hilfe, wir haben zugesagt, aber noch nicht geantwortet
#   hilfe_zugesagt       wir sind unterwegs zu einem anderen Trupp
#   rauch_da             er hat eine Rauchgranate
#   trupp_igel           der Trupp igelt sich ein (von allen Seiten bedroht, seit 0.34.41)
#   trupp_engstelle      der Trupp haelt eine Engstelle gegen eine Horde (seit 0.34.41)
# Seit 0.34.40, Bergen (Testmodus: VERWUNDETE BOTS):
#   bei_kamerad          er steht neben dem, den er bergen soll
#   kamerad_gedeckt      der Liegende ist vor jedem bekannten Gegner gedeckt (oder im Rauch)
#   ich_berge            er ist einer der Retter (im Trupp bestimmt, sonst die zwei naechsten)
#   bergen_sicher        es scheint sicher genug, ihn zu holen
#   bergung_laeuft       einer zieht ihn gerade oder hilft ihm auf
#   anderer_zieht        ein anderer zieht ihn schon (seit 0.34.50) - ziehen kann nur einer
#   bergung_halten       im Trupp wird einer geborgen, und er ist nicht der Retter: er
#                        bleibt in Deckung (seit 0.34.53) - kein Nachsetzen, Flankieren, Suchen
#   hilft_schon          er hilft ihm schon auf (seit 0.34.55): nicht mehr zum Ziehen
#   fast_fertig          ... und hat schon VERWUNDET["zu_ende"] geschafft: er bringt es zu Ende
#   anderer_hilft        ein anderer hilft ihm schon auf (seit 0.34.55): nicht wegziehen, mithelfen
#   darf_fliehen         das Spiel erlaubt Flucht in Panik (seit 0.34.56): Kampagne ja, Runden nein
#                        umschwenken - jede Unterbrechung kostet Fortschritt
#   geschwaecht          er wurde wiederbelebt: weniger Leben, langsamer, zielt schlechter
# Ziel-Fakten, die nie wahr werden, sondern nur das Ziel benennen:
#   feind_besiegt, kamerad_geschuetzt, sturm_fertig, gruppe_getroffen, sicher,
#   welle_fertig, bei_hilfe, gemeldet, geantwortet, rauch_liegt
# Zahlen: leben (0 bis 1), abstand (zum Gegner, sonst 9999),
#   stress (0 bis 1), zuversicht (0 panisch ... 0.5 neutral ... 1 heldenhaft)
# Merkmale (je Bot gewuerfelt): mut, disziplin, drill
#
# Rang (seit 0.34.38): 3 Granate, 2 Panik, 1 was keinen Aufschub duldet,
# 0 alles andere. Ein hoeherer Rang unterbricht sofort; ein gleicher erst,
# wenn das laufende Ziel `bindung` Sekunden lang verfolgt wurde. Gemeldet:
# "Sie sollen etwas mehr Commitment zu ihren Aufgaben zeigen. Sonst ist es
# irgendwann quasi unmoeglich, Gegner zu toeten, weil diese direkt auf
# alles reagieren."

# ── Ziele ─────────────────────────────────────────────────────────────

[ziel granate_ausweichen]
will: !granate_nah
rang: 3
dringlich: 120 * granate_nah
info: Eine Granate liegt neben ihm: weg - vor allem anderen.

[ziel panik]
will: sicher
rang: 2
dringlich: 100 * panik
info: Panik: nur noch weg - oder, wer schon in Deckung ist, in ihre beste Ecke. Feuerschutz, Sturm, Kreis - alles egal.

[ziel flanke_sichern]
will: !flankiert
# Seit 0.34.50. Gemeldet: "die Bots sollen wirklich darauf aufpassen, dass
# sie nie von der Flanke angeschossen werden ... sich eine Position suchen,
# wo sie beide Gegner noch sehen koennen." Rang 1: vor dem Feuerkampf.
rang: flankiert
dringlich: 95 * flankiert * (1 - 0.6 * ziel_fast_tot)
bindung: 1.5
info: Gegner von zwei Seiten: dorthin, wo alle vor ihm stehen (oder er vor jedem gedeckt ist).

[ziel ueberleben]
will: in_deckung, beruhigt
# Nicht mehr jede Kugel (0.34.37: "unterdrueckt" allein reichte), sondern
# der Stress, der sich aufbaut. Wer das MG am Laufen hat oder den Gegner
# gleich erledigt, bleibt draussen. Ganz knapp am Ende hat er Rang 1.
# Der Anfuehrer will mehr als die anderen ueberleben (0.34.39).
rang: (stress > 0.85) + angeschlagen * getroffen * (leben < 0.25)
dringlich: (110 * stress + 45 * angeschlagen * getroffen) * (1.3 - mut) * (1.4 - 0.4 * zuversicht) * (1 - 0.6 * eingeschossen * waffe_mg) * (1 - 0.5 * ziel_fast_tot) * (1 + 0.35 * ist_fuehrer + 0.3 * geschwaecht)
bindung: 1.0 + 2 * disziplin
info: Gestresst oder schwer getroffen: in Deckung, bis er sich gefangen hat.

[ziel nachladen]
will: !magazin_leer
rang: 1
dringlich: 85 * magazin_leer * munition_da
bindung: 0.5
info: Ein leeres Magazin muss voll, bevor er weiterkaempft.

[ziel kamerad_bergen]
will: !kamerad_liegt
# Seit 0.34.40 (vorher: aufhelfen, nur wenn kein Gegner dicht dran war).
# Nur wer dran ist, und nur, wenn es sicher genug scheint - erst in
# Deckung ziehen, dann aufhelfen; die anderen decken (feuerschutz).
# Ist es sicher (oft erst durch den Rauch), hat es Rang 1 - so lange haelt
# der Rauch nicht, dass der Retter erst seinen Feuerstoss zu Ende bringt.
# Fast fertig (seit 0.34.55): Rang 3 - nur die Granate (120) geht dann noch
# vor; Panik, Ueberleben und Flanke warten die eine Sekunde. Vorher liess er
# bei neun Zehnteln los, um in Deckung zu gehen, und der Fortschritt verfiel.
rang: kamerad_liegt * ich_berge * bergen_sicher * (1 + 2 * fast_fertig)
dringlich: kamerad_liegt * ich_berge * bergen_sicher * (72 + 20 * disziplin) * (1 - 0.5 * geschwaecht)
bindung: 3
info: Einen gefallenen Kameraden in Sicherheit ziehen und aufhelfen.

[ziel verstaerkung_rufen]
will: hilfe_gerufen
# Gemeldet: "wenn das eine Team sagt, wir sind in der Unterzahl ... dann
# sollen sie nach Hilfe rufen." Nur wer fuehrt und ein Funkgeraet hat -
# ohne das ist es einfach kein Ziel, das er erreichen kann.
rang: hat_funk * ist_fuehrer * trupp_unterlegen
dringlich: 92 * hat_funk * ist_fuehrer * trupp_unterlegen
bindung: 3
info: Der Anfuehrer funkt um Hilfe, wenn sein Trupp unterlegen ist.

[ziel funk_antworten]
will: geantwortet
# Ueber dem Kampf (80): ein "Verstanden, kommen!" ist kurz, und der andere
# wartet darauf.
dringlich: 88 * hat_funk * ist_fuehrer * antwort_offen
bindung: 3
info: Ein anderer rief, wir kommen - das sagt er ihm auch.

[ziel lage_melden]
will: gemeldet
dringlich: 46 * hat_funk * ist_fuehrer * lage_neu * (1 - feind_sichtbar * 0.5)
bindung: 3
info: Hat sich viel geaendert, meldet der Anfuehrer es den anderen Trupps.

[ziel rauch_legen]
will: rauch_liegt
# Der Anfuehrer deckt den Rueckzug mit seinem Rauch - und das Bergen: liegt
# einer im Blick der Gegner, macht erst der Rauch es sicher genug.
# Rang 1 beim Bergen (seit 0.34.54): sonst kaeme der Feuerschutz vor dem Rauch.
rang: rauch_da * ist_fuehrer * kamerad_liegt * feind_bekannt
dringlich: 92 * rauch_da * ist_fuehrer * min(1, trupp_rueckzug + kamerad_liegt) * feind_bekannt
bindung: 1
info: Rauch genau dorthin, wo er gebraucht wird: auf den Liegenden, beim Rueckzug auf den Trupp.

[ziel rueckzug]
will: welle_fertig
# Der Trupp zieht sich in Wellen zurueck. Wer wenig Disziplin hat, macht
# eher sein eigenes Ding (oder rennt in Panik).
dringlich: 95 * trupp_rueckzug * (0.4 + disziplin)
bindung: 2
info: Rueckzug in Wellen: die Haelfte zieht, die andere deckt.

[ziel druck]
will: welle_fertig
dringlich: 86 * trupp_druck * (0.45 + mut) * (1 - 0.6 * ist_fuehrer) * (1 - 0.7 * geschwaecht) + 90 * flanke_rolle
bindung: 2
info: Druck in Wellen, die Aeusseren flankieren.

[ziel zu_hilfe]
will: bei_hilfe
# "dann toeten wir die und kommen dann sofort zur Hilfe": erst die eigenen.
dringlich: 74 * hilfe_zugesagt * (1 - feind_sichtbar) * (1 - 0.6 * feind_bekannt)
bindung: 2
info: Zu einem Trupp, der um Hilfe gerufen hat - wenn die eigenen Gegner erledigt sind.

[ziel sektor_halten]
will: sektor_gesichert
# Eingeigelt oder an der Engstelle: an seinem Platz bleiben und seine
# Richtung im Blick behalten - nicht flankieren, nicht nachsetzen, nicht
# suchen gehen. Wer auftaucht, ist fuer feind_ausschalten (80).
dringlich: 72 * min(1, trupp_igel + trupp_engstelle) * (1 - feind_sichtbar)
bindung: 1.5
info: Im Igel oder an der Engstelle seinen Sektor halten.

[ziel trupp_halten]
will: in_formation
# An seiner Stellung bleiben: im Kampf seine Bahn, sonst die Linie. Wer
# allein ist (keiner in Sprechweite), will dringend zurueck; der Anfuehrer
# haelt seinen Platz in der Mitte lieber als die anderen.
dringlich: im_trupp * (1 - in_formation) * (34 + 22 * ist_fuehrer + 34 * allein + 20 * trupp_halten)
bindung: 1
info: Im Trupp an seinem Platz bleiben - in Sprechweite.

[ziel stuermen]
will: sturm_fertig
dringlich: 60 * sturm_plan * (1 - 0.8 * geschwaecht)
bindung: 2
info: Die Mannschaft stuermt gemeinsam (bottaktik.stuerme_fuehren).

[ziel kreis]
will: im_kreis
# Ueber feind_ausschalten (bis 80): draussen geht der Kreis vor, auch wenn
# er einen Gegner sieht - zum_kreis kaempft auf dem Weg. Mit 57 blieb er
# stehen und kaempfte, wo er war (gesehen in der Aufnahme, 0.34.31).
# Ueberleben und Nachladen gehen weiter vor.
dringlich: 82 * zone_aktiv
bindung: 1.5
info: HUEGEL: Der Kreis geht vor - gekaempft wird auf dem Weg dorthin.

[ziel gruppe_bekaempfen]
will: gruppe_getroffen
# Mehrere beisammen: Granate hinein oder mit Dauerfeuer darueber - statt
# einen nach dem anderen (seit 0.34.38).
dringlich: 88 * feinde_gruppe * feind_sichtbar
bindung: 1.5 + disziplin
info: Mehrere Gegner beisammen: was viele zugleich trifft.

[ziel feuerschutz]
will: kamerad_geschuetzt
# Waehrend einer geborgen wird (seit 0.34.54), ist das die Aufgabe aller
# anderen - Rang 1, vor Welle, Funk und Feuerkampf: aus der Deckung den
# Gegner niederhalten, der den Retter treffen koennte. Gemeldet: "dann kommt
# ein anderer raus und ich kann ihn einfach wieder anschiessen, werde dabei
# nicht beschossen." Nachladen (85), Flanke (95), Panik und Granate gehen vor.
rang: bergung_halten * feind_bekannt
dringlich: 52 * kamerad_braucht_schutz * feind_bekannt * (1 - feind_sichtbar) * (0.5 + disziplin) * (1 - bergung_halten) + 80 * bergung_halten * feind_bekannt
bindung: 1.5
info: Ein Kamerad braucht Schutz und der Gegner ist in Deckung: niederhalten.

[ziel nachsetzen]
will: feind_sichtbar
# Uebermut: zuversichtlich und wenig Disziplin - er jagt dem nach, der
# eben verschwand, statt in Deckung zu warten. Ein Anfuehrer tut das nicht.
dringlich: 70 * (zuversicht > 0.7) * (1.2 - disziplin) * feind_bekannt * (1 - feind_sichtbar) * (1 - ist_fuehrer) * (1 - geschwaecht) * (1 - min(1, trupp_igel + trupp_engstelle))
bindung: 2
info: Uebermut: dem Gegner hinterher, ohne Deckung.

[ziel feind_ausschalten]
will: feind_besiegt
# Wer fast erledigt ist, wird fertig gemacht (0.34.38) - nur nicht im
# HUEGEL: dort geht der Kreis vor, und zum_kreis kaempft ohnehin auf dem Weg.
dringlich: 50 * feind_sichtbar + 30 * feind_bekannt + 25 * ziel_fast_tot * (1 - zone_aktiv)
# Kuerzer gebunden seit 0.34.43: sonst blieb er im Freien beim Angriff,
# obwohl eine Deckung daneben stand.
bindung: 0.6 + 0.8 * disziplin
info: Den Gegner, den er sieht oder eben sah, ausschalten.

[ziel suchen]
will: feind_sichtbar
# Waehrend einer geborgen wird, gar nicht erst wollen (seit 0.34.55): die
# Aktion suchen braucht dann !bergung_halten, und das Ziel lief jeden
# Denkschritt vergeblich durch den Planer.
dringlich: (10 + 15 * feind_gehoert) * (1 - bergung_halten)
info: Nichts los: Gegner suchen - wo er einen hoerte, sonst auf der Karte.

[aktion ausweichen]
braucht: granate_nah
macht: !granate_nah
kosten: 1
tut: ausweichen

# Panik (seit 0.34.48): bleiben, eine bessere Deckung weiter hinten, oder
# ganz weg - je nach Leben, Zuversicht und dem, was der Trupp tut. Gesund,
# zuversichtlich, der Trupp rueckt vor: er bleibt. Angeschlagen: in eine
# bessere Deckung. Der Trupp zieht sich zurueck und er ist schwer
# getroffen: er rennt.
[aktion fliehen]
# Nur, wo das Spiel es erlaubt (seit 0.34.56): in Runden nicht - dort sah
# Wegrennen wie ein Fehler aus. Die Kampagne erlaubt es.
braucht: darf_fliehen
macht: sicher
# Im Trupp flieht keiner allein, solange der Trupp nicht zurueckgeht
# (seit 0.34.50) - dann eher bleiben oder eine Deckung beim Trupp.
kosten: 0.8 + 1.5 * leben + 0.8 * trupp_druck - 0.6 * trupp_rueckzug + 1.2 * im_trupp * (1 - trupp_rueckzug)
tut: fliehen

[aktion umstellen]
macht: !flankiert
kosten: 1
tut: umstellen
info: Hinueber an einen Platz, von dem aus alle Gegner vor ihm sind - im Gehen auf den, der auf ihn schiesst.

[aktion verkriechen]
braucht: in_deckung
macht: sicher
kosten: 0.5 + 1.2 * (1 - leben) + 0.6 * (1 - zuversicht) + 0.8 * trupp_rueckzug
tut: verkriechen
info: Panik in der Deckung: dort bleiben, in der besten Ecke, die gedeckt erreichbar ist.

[aktion deckung_wechseln]
macht: sicher
kosten: 0.9 + 0.3 * leben + 0.6 * trupp_druck
tut: deckung_wechseln
info: Panik: in eine bessere Deckung weiter weg vom Gegner, auf einem Weg, der nicht an ihm vorbeifuehrt.

[aktion angreifen]
braucht: feind_sichtbar, linie_frei, munition_da
macht: feind_besiegt, gruppe_getroffen
# Seit 0.34.43 teurer als Deckung + aus_deckung_feuern (2 + mut), ausser
# fuer die Mutigen, fuer den, der das MG am Laufen hat, und wenn der
# Gegner gleich fertig ist: "Sie sollen schon priorisieren, in Deckung zu
# gehen."
kosten: 1.5 + 4 * (1 - mut) * (1 - eingeschossen * waffe_mg) * (1 - ziel_fast_tot)
tut: kaempfen
info: Kaempfen wie die klassische KI: Abstand, Ausweichen, Feuerstoesse, Granate.

[aktion aus_deckung_feuern]
braucht: in_deckung, feind_bekannt, munition_da
# Auch Feuerschutz (seit 0.34.50): wer den Rettern Deckung gibt, tut es aus
# der Deckung - vorher standen dafuer alle im Freien (niederhalten).
macht: feind_besiegt, gruppe_getroffen, kamerad_geschuetzt
kosten: 1
tut: aus_deckung
info: Aus der Deckung kaempfen: drin warten, zur Seite herauslehnen, ein Stoss, zurueck.

[aktion dauerfeuer]
braucht: feind_sichtbar, linie_frei, waffe_auto, munition_da
macht: gruppe_getroffen
kosten: 0.8 - 0.3 * waffe_mg
tut: dauerfeuer
funk: gruppe
info: Stehen bleiben und lange ueber die Gruppe halten.

[aktion granate_werfen]
braucht: feind_sichtbar, granate_da
macht: gruppe_getroffen
kosten: 0.5
tut: werfen

[aktion flankieren]
braucht: feind_bekannt, !linie_frei, flanke_frei, munition_da, !bergung_halten
macht: feind_sichtbar, linie_frei
kosten: 3 - mut + 5 * ist_fuehrer + 4 * geschwaecht + 8 * min(1, trupp_igel + trupp_engstelle)
tut: flankieren
funk: flanke

[aktion niederhalten]
braucht: feind_bekannt, munition_da
macht: kamerad_geschuetzt, feind_besiegt
kosten: 4
tut: niederhalten
funk: druck
info: Neben die Deckung schiessen. Fuer den Gegner nur zweite Wahl nach der Flanke.

[aktion nahkampf]
braucht: feind_sichtbar, !munition_da
macht: feind_besiegt
kosten: 2
tut: nahkampf
funk: leer

[aktion suchen]
# Waehrend einer geborgen wird, geht keiner suchen (seit 0.34.53): die
# anderen geben aus der Deckung Feuerschutz, statt mit hinaus zu laufen.
braucht: !bergung_halten
macht: feind_sichtbar, linie_frei, feind_bekannt
kosten: 5
tut: suchen

[aktion kamerad_ziehen]
# Wer schon aufhilft, zieht nicht mehr (seit 0.34.55): vorher plante er
# mitten im Aufhelfen das Ziehen fuer billiger, liess los, das Ziehen
# scheiterte, und er fing von vorn an - gemessen bis zu viermal.
braucht: bei_kamerad, !kamerad_gedeckt, !anderer_zieht, !hilft_schon, !anderer_hilft
macht: kamerad_gedeckt
kosten: 2
tut: ziehen

tut: helfen
